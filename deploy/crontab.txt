# cPanel › Cron Jobs. Paths match the install in ~/nexopay3 with Python 3.11.
# Every minute: release expired holds, open/close events, reconcile payments, send ticket emails.
* * * * * cd $HOME/nexopay3 && /usr/bin/flock -n $HOME/.nexopay3-cron.lock $HOME/virtualenv/nexopay3/3.11/bin/python manage.py nexopay_cron >> $HOME/nexopay3/logs/cron.log 2>&1

# 03:15 every night: database backup (to R2 when configured, otherwise ~/nexopay3/private_media/backups), keeps 14 days.
15 3 * * * cd $HOME/nexopay3 && $HOME/virtualenv/nexopay3/3.11/bin/python manage.py backup_db --keep 14 >> $HOME/nexopay3/logs/backup.log 2>&1

# Mondays 04:30: prune finished background-task rows.
30 4 * * 1 cd $HOME/nexopay3 && $HOME/virtualenv/nexopay3/3.11/bin/python manage.py prune_db_task_results >> $HOME/nexopay3/logs/cron.log 2>&1
